	\begin{frame}{Introduction}
		
		\small
		
		\begin{block}{Context}
			 It is more expensive to sign in a new client than keeping an existing one. Losing existing customers reduces revenue, weakens customer loyalty and increases the cost of acquiring new clients. In microfinance institutions, customer retention is important due to the dependence on long-term customer relationships and repeat borrowing behaviour.
		\end{block}
		
		\vspace{0.0002cm}
		
		\begin{columns}
			
			%================================================
			% LEFT COLUMN
			%================================================
			\column{0.48\textwidth}
			
			\begin{alertblock}{Key Issues}
				\begin{itemize}
					\item Increasing customer attrition
					\item Reduced customer lifetime value
					\item High customer replacement costs
					\item Limited predictive retention systems
				\end{itemize}
			\end{alertblock}
			
			%================================================
			% RIGHT COLUMN
			%================================================
			\column{0.48\textwidth}
			
			\begin{exampleblock}{Focus}
				\begin{itemize}
					\item Analyze factors influencing churn
					\item Train a Random Forest model
					\item Predict customer exit behavior
					\item Deploy a prediction system
				\end{itemize}
			\end{exampleblock}
			
		\end{columns}
		
		\vspace{0.0003cm}
		
		\begin{block}{Why Random Forest?}
			Random Forest algorithm provides high predictive accuracy, handles non-linear customer behaviour patterns, reduces overfitting and produces interpretable feature importance measures useful for managerial decision making.
		\end{block}
		
	\end{frame}

	% BACKGROUND TO THE STUDY
	%================================================
	\begin{frame}{Background to the Study}
		
		\small
		
		\begin{block}{Overview}
		Studies show that retaining existing customers is significantly less costly than acquiring new ones (Kotler \& Keller, 2016). Consequently, organizations are increasingly adopting predictive analytics and machine learning techniques to improve customer retention strategies.
		\end{block}
		
		\vspace{0.0000002cm}
		
		\begin{columns}
			
			%================================================
			% LEFT COLUMN
			%================================================
			\column{0.48\textwidth}
			
			\begin{exampleblock}{Microfinance}
				Microfinance institutions rely on long-term customer relationships, repeat borrowing and trust. High churn rates negatively affect profitability, customer lifetime value, and institutional sustainability.
			\end{exampleblock}
			
			%================================================
			% RIGHT COLUMN
			%================================================
			\column{0.48\textwidth}
			
			\begin{exampleblock}{Machine Learning}
				Recent studies indicate that ML models such as XGBoost, Random Forest and Neural Nets are best in churn prediction due to their ability to model nonlinear behaviour patterns (Huang et al., 2012).
			\end{exampleblock}
			
		\end{columns}
		
		\vspace{0.002cm}
		
		

\begin{block}{Major Motivation}
	Most local customer churn studies focus on identifying churn determinants and predictive analytics without deploying practical decision-support systems. This study combines customer retention analytics, ML and web-based deployment to support customer retention at KCI Management Consultants.
\end{block}

	\vspace{0.003cm}
			
	\end{frame}

\begin{frame}{Problem Statement}
	
	\small
	
	\begin{alertblock}{Challenges}
		\begin{itemize}
			\item KCI Management Consultants faces the risk of losing customers through churn.
			
			\item Customer attrition reduces revenue, customer loyalty and long-term profitability.
			
			\item Existing retention approaches are largely reactive rather than predictive.

% LITERATURE REVIEW 
%================================================
\begin{frame}{Literature Review}
	
	\small


		
\begin{exampleblock}{Previous Studies in Zimbabwe}
	\begin{itemize}
		
		\item Revesai et al. (2022) studied customer churn in Zimbabwean banks using Logistic Regression, Decision Trees, Random Forest, XGBoost and Deep Learning models. Neural Networks(87\% accuracy) outperfomed all models followed by random forests(86.5\% accuracy), but the study had limited feature coverage and lacked deployment of a practical system.
		
		\item Chiurunge et al. (2023) developed an Artificial Neural Network model for churn prediction: Case study of Econet Wireless Zimbabwe. The study focused mainly on predictive accuracy without interpretable feature analysis and deployment integration.
		
		\item Vhimisai et al. (2024) examined customer switching behaviour in the Zimbabwean Mobile Telecommunication Industry using explanatory statistical approaches. The study did not implement machine learning models for churn prediction.
		
		\item Ahmed et al. (2024) compared Logistic Regression, Random Forest, GBM and SVM in retail banking churn prediction and found ensemble methods highly effective all with more than 80\% accuracy, although the study did not focus on deploying systems within microfinance institutions.
		
	\end{itemize}
\end{exampleblock}

% MODEL DEVELOPMENT

\begin{frame}{Con't}
	
	\small
	
	\begin{block}{Random Forest Classifier}
		The study will employ the Random Forest algorithm to classify customers according to their probability of churn. Random Forest is ensemble learning algorithm constructs multiple decision trees and combines their predictions to improve classification accuracy.
	\end{block}
	
	\vspace{0.2cm}
	
	\begin{exampleblock}{Random Forest Prediction}
		\[
		\hat{Y} = \text{Majority Vote}(T_1,T_2,T_3,\dots,T_n)
		\]
	\end{exampleblock}
	
	\vspace{0.2cm}

		
		\begin{exampleblock}{Model Optimization}
			\begin{itemize}
				\item Hyperparameter tuning
				
				\item Cross-validation
				
				\item Tree depth optimization
				
				\item Feature importance analysis
			\end{itemize}
		\end{exampleblock}
		

	
\end{frame}




\begin{frame}{Con't}
	
	\small
	
	\begin{block}{Evaluation Metrics}
		The performance of the Random Forest model will be evaluated using classification metrics derived from the confusion matrix.
	\end{block}
	
	\vspace{0.2cm}
	
	\begin{exampleblock}{Confusion Matrix}
		\[
		\begin{bmatrix}
			TP & FP \\
			FN & TN
		\end{bmatrix}
		\]
	\end{exampleblock}
	
	\vspace{0.2cm}
	
	\begin{columns}
		
		%================================================
		% LEFT COLUMN
		%================================================
		\column{0.48\textwidth}
		
		\begin{alertblock}{Performance Metrics}
			\[
			Accuracy = \frac{TP+TN}{TP+TN+FP+FN}
			\]
			
			\[
			Precision = \frac{TP}{TP+FP}
			\]
			
			\[
			Recall = \frac{TP}{TP+FN}
			\]
		\end{alertblock}
		
		%================================================
		% RIGHT COLUMN
		%================================================
		\column{0.48\textwidth}
		
		\begin{exampleblock}{ }
			\[
			F1 = 2 \times \frac{Precision \times Recall}{Precision+Recall}
			\]
			
			\[
		 	ROC\text{ }AUC
			\]
		
		\end{exampleblock}
		
	\end{columns}
	
\end{frame}

\begin{frame}{Definition of Key Terms}
	
	\small
	
	\begin{block}{Operational Definitions}
	\end{block}
	
	\vspace{0.2cm}
	
	\begin{description}
		
		\item[\textbf{Customer Attrition}]
		The gradual loss of customers over time due to termination of business relationships with an institution.
		
		\vspace{0.1cm}
		
		\item[\textbf{Customer Retention}]
		Strategies and processes used by organizations to maintain long-term customer relationships and reduce customer loss.
		
		\vspace{0.1cm}
		
		\item[\textbf{Customer Churn}]
		The process in which a customer discontinues their relationship with a financial institution.
		
		\vspace{0.1cm}
		
		\item[\textbf{Random Forest}]
		An ensemble machine learning algorithm that combines multiple decision trees for classification and prediction.
		
		\vspace{0.1cm}
		
		\item[\textbf{Predictive Analytics}]
		The use of statistical and machine learning techniques to predict future outcomes using historical data.
		
		\vspace{0.1cm}
		
		\item[\textbf{Machine Learning}]
		A branch of artificial intelligence that enables systems to learn patterns from data and make predictions.
		
		\vspace{0.1cm}
		
		\item[\textbf{Web-Based Deployment}]
		Implementation of a trained machine learning model within an online application for real-time usage.
		
	\end{description}
